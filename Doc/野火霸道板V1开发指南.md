# STM32F103ZET6 野火霸道板 V1 开发指南

## 一、硬件配置总览

### 1.1 LED 配置

| LED颜色 | GPIO | 低电平点亮 | 说明 |
|---------|------|----------|------|
| **绿灯** | PB5 | ✅ 是 | 低电平点亮（灌电流） |
| **黄灯** | PA12 | ✅ 是 | 低电平点亮 |
| **红灯** | PA11 | ✅ 是 | 低电平点亮，但可能被SWD占用 |

**重要**：
- PB5 不被调试接口占用，推荐使用
- PA11 在某些情况下可能被 SWD 调试占用
- PA12 正常使用

**LED 代码示例**：
```c
// 配置LED（PB5）
RCC->APB2ENR |= RCC_APB2ENR_IOPBEN;
GPIOB->CRL &= ~(GPIO_CRL_CNF5 | GPIO_CRL_MODE5);
GPIOB->CRL |= GPIO_CRL_MODE5_0 | GPIO_CRL_MODE5_1;  // 推挽输出 50MHz

// 点亮LED（低电平）
GPIOB->BSRR = GPIO_BSRR_BR5;  // 亮

// 熄灭LED（高电平）
GPIOB->BSRR = GPIO_BSRR_BS5;  // 灭
```

### 1.2 按键配置

| 按键 | GPIO | 有效电平 | 说明 |
|------|------|----------|------|
| **KEY1** | PA0 | 低电平 | 带唤醒功能，外部上拉 |
| **KEY2** | PC13 | 低电平 | 有特殊限制 |

**重要**：
- KEY1 (PA0) 推荐使用，配置简单
- KEY2 (PC13) 在 STM32F103 上有限制，不推荐优先使用
- 按键需要外部上拉电阻或配置内部上拉

**按键代码示例（推荐方式）**：
```c
// 配置KEY1（PA0）- 浮空输入
RCC->APB2ENR |= RCC_APB2ENR_IOPAEN;
GPIOA->CRL &= ~(GPIO_CRL_CNF0 | GPIO_CRL_MODE0);
GPIOA->CRL |= GPIO_CRL_CNF0_0;  // 浮空输入

// 下降沿检测
uint8_t keyPrev = 1;
while(1) {
    uint8_t keyNow = (GPIOA->IDR & GPIO_IDR_IDR0);
    if(keyPrev && !keyNow) {  // 下降沿
        // 按键按下处理
    }
    keyPrev = keyNow;
}
```

### 1.3 串口配置

| 串口 | TX | RX | 波特率 | 用途 |
|------|----|----|--------|------|
| **USART1** | PA9 | PA10 | 115200 | 调试/ESP8266 |
| **USART3** | PB10 | PB11 | 115200 | ESP8266专用 |

**重要**：
- USART1 可用于调试或 ESP8266 通信
- USART3 推荐用于 ESP8266，释放 USART1 用于调试
- 两个串口引脚配置相同

**串口代码示例**：
```c
// 配置USART1（PA9/PA10）
RCC->APB2ENR |= RCC_APB2ENR_IOPAEN | RCC_APB2ENR_USART1EN;

// PA9: TX 复用推挽输出
GPIOA->CRH &= ~(GPIO_CRH_CNF9 | GPIO_CRH_MODE9);
GPIOA->CRH |= GPIO_CRH_CNF9_1 | GPIO_CRH_MODE9_0 | GPIO_CRH_MODE9_1;

// PA10: RX 浮空输入
GPIOA->CRH &= ~(GPIO_CRH_CNF10 | GPIO_CRH_MODE10);
GPIOA->CRH |= GPIO_CRH_CNF10_0;

// 波特率 115200 @ 72MHz
USART1->BRR = 0x0271;  // 或 0x1D4C
USART1->CR1 |= USART_CR1_TE | USART_CR1_UE;
```

### 1.4 系统时钟配置

**重要**：
- 野火霸道板使用 **8MHz 外部晶振 (HSE)**
- 系统默认配置为 **72MHz**
- 使用 HSI (8MHz) 时，串口波特率计算不同

**HSE 配置示例（推荐）**：
```c
// 使用HSE 8MHz晶振
RCC->CR |= RCC_CR_HSEON;
while(!(RCC->CR & RCC_CR_HSERDY));  // 等待HSE就绪

RCC->CFGR &= ~RCC_CFGR_SW;
RCC->CFGR |= RCC_CFGR_SW_HSE;  // HSE作为系统时钟

// PLL: 8MHz * 9 = 72MHz
RCC->CFGR |= RCC_CFGR_PLLMULL9;
RCC->CR |= RCC_CR_PLLON;
while(!(RCC->CR & RCC_CR_PLLRDY));

RCC->CFGR &= ~RCC_CFGR_SW;
RCC->CFGR |= RCC_CFGR_SW_PLL;  // PLL作为系统时钟
```

## 五、ESP8266 WiFi 模块实现

### 5.1 野火霸道板V1 板载ESP8266

野火霸道板V1 **板载ESP8266模块**，使用 **USART3 (PB10/PB11)**：

| 板载接口 | STM32引脚 | 说明 |
|---------|----------|------|
| ESP8266_TX | PB11 (USART3_RX) | STM32接收 |
| ESP8266_RX | PB10 (USART3_TX) | STM32发送 |

### 5.2 串口配置（野火霸道板V1 - USART3）

```c
// USART3 - 板载ESP8266通信 (PB10/PB11)
#define ESP8266_USART   USART3
#define ESP8266_BAUD    115200

void ESP8266_Init(void) {
    GPIO_InitTypeDef GPIO_InitStructure;
    USART_InitTypeDef USART_InitStructure;
    NVIC_InitTypeDef NVIC_InitStructure;
    
    // 使能时钟
    RCC_APB2PeriphClockCmd(RCC_APB2Periph_GPIOB, ENABLE);
    RCC_APB1PeriphClockCmd(RCC_APB1Periph_USART3, ENABLE);
    
    // PB10: TX (复用推挽输出)
    GPIO_InitStructure.GPIO_Pin = GPIO_Pin_10;
    GPIO_InitStructure.GPIO_Mode = GPIO_Mode_AF_PP;
    GPIO_InitStructure.GPIO_Speed = GPIO_Speed_50MHz;
    GPIO_Init(GPIOB, &GPIO_InitStructure);
    
    // PB11: RX (浮空输入)
    GPIO_InitStructure.GPIO_Pin = GPIO_Pin_11;
    GPIO_InitStructure.GPIO_Mode = GPIO_Mode_IN_FLOATING;
    GPIO_Init(GPIOB, &GPIO_InitStructure);
    
    // USART配置 115200, 8N1
    USART_InitStructure.USART_BaudRate = 115200;
    USART_InitStructure.USART_WordLength = USART_WordLength_8b;
    USART_InitStructure.USART_StopBits = USART_StopBits_1;
    USART_InitStructure.USART_Parity = USART_Parity_No;
    USART_InitStructure.USART_HardwareFlowControl = USART_HardwareFlowControl_None;
    USART_InitStructure.USART_Mode = USART_Mode_Rx | USART_Mode_Tx;
    USART_Init(USART3, &USART_InitStructure);
    
    // 使能接收中断
    USART_ITConfig(USART3, USART_IT_RXNE, ENABLE);
    
    // NVIC配置
    NVIC_InitStructure.NVIC_IRQChannel = USART3_IRQn;
    NVIC_InitStructure.NVIC_IRQChannelPreemptionPriority = 2;
    NVIC_InitStructure.NVIC_IRQChannelSubPriority = 2;
    NVIC_InitStructure.NVIC_IRQChannelCmd = ENABLE;
    NVIC_Init(&NVIC_InitStructure);
    
    USART_Cmd(USART3, ENABLE);
}
```

### 5.3 接收缓冲区设计

```c
// 重要：缓冲区需要足够大，防止MQTT包溢出
#define ESP8266_BUF_SIZE     256   // 最小256字节
#define ESP8266_RX_BUF_SIZE   512  // 接收缓冲区

unsigned char esp8266_buf[ESP8266_BUF_SIZE];
unsigned short esp8266_cnt = 0, esp8266_cntPre = 0;
uint8_t esp8266_rx_buf[ESP8266_RX_BUF_SIZE];
uint16_t esp8266_rx_index = 0;

// USART中断处理
void USARTx_IRQHandler(void) {
    if(USART_GetITStatus(USARTx, USART_IT_RXNE) != RESET) {
        uint8_t ch = USART_ReceiveData(USARTx);
        
        // OneNET兼容缓冲区（防止溢出）
        if(esp8266_cnt < ESP8266_BUF_SIZE - 1) {
            esp8266_buf[esp8266_cnt++] = ch;
        }
        
        // 增强版缓冲区
        if(esp8266_rx_index < ESP8266_RX_BUF_SIZE - 1) {
            esp8266_rx_buf[esp8266_rx_index++] = ch;
        }
        
        USART_ClearITPendingBit(USARTx, USART_IT_RXNE);
    }
}
```

### 5.4 AT指令发送函数

```c
#define REV_OK    0  // 接收完成标志
#define REV_WAIT  1  // 接收未完成标志

// 清空缓冲区
void ESP8266_Clear(void) {
    memset(esp8266_buf, 0, sizeof(esp8266_buf));
    esp8266_cnt = 0;
    esp8266_cntPre = 0;
}

// 等待接收完成
_Bool ESP8266_WaitRecive(void) {
    if(esp8266_cnt == 0)
        return REV_WAIT;
    
    if(esp8266_cnt == esp8266_cntPre) {
        esp8266_cnt = 0;
        return REV_OK;
    }
    
    esp8266_cntPre = esp8266_cnt;
    return REV_WAIT;
}

// 发送AT指令
_Bool ESP8266_SendCmd(char *cmd, char *res) {
    unsigned char timeOut = 200;
    
    ESP8266_Clear();
    ESP8266_SendString(cmd);
    ESP8266_SendString("\r\n");
    
    while(timeOut--) {
        if(ESP8266_WaitRecive() == REV_OK) {
            if(strstr((const char *)esp8266_buf, res) != NULL) {
                ESP8266_Clear();
                return 0;  // 成功
            }
        }
        Delay_ms(10);
    }
    return 1;  // 失败
}

// 发送数据
void ESP8266_SendData(unsigned char *data, unsigned short len) {
    char cmdBuf[32];
    
    ESP8266_Clear();
    sprintf(cmdBuf, "AT+CIPSEND=%d", len);
    
    if(!ESP8266_SendCmd(cmdBuf, ">")) {
        ESP8266_SendDataArray(data, len);
    }
}
```

### 5.5 WiFi连接流程

```c
uint8_t ESP8266_ConnectWiFi(char *ssid, char *password) {
    char cmd[100];
    
    // 1. 设置Station模式
    ESP8266_SendCmd("AT+CWMODE=1", "OK");
    Delay_ms(100);
    
    // 2. 先断开之前的连接
    ESP8266_SendCmd("AT+CWQAP", "OK", 2000);
    Delay_ms(500);
    
    // 3. 连接WiFi
    sprintf(cmd, "AT+CWJAP=\"%s\",\"%s\"", ssid, password);
    ESP8266_ClearBuffer();
    ESP8266_SendString(cmd);
    ESP8266_SendString("\r\n");
    
    // 4. 等待连接成功（最多15秒）
    uint32_t tick = 0;
    while(tick < 15000) {
        if(strstr((char*)esp8266_rx_buf, "WIFI GOT IP") != NULL) {
            return 1;  // 成功
        }
        if(strstr((char*)esp8266_rx_buf, "FAIL") != NULL) {
            return 0;  // 失败
        }
        Delay_ms(10);
        tick += 10;
    }
    return 0;
}
```

### 6.3 设备连接函数

```c
_Bool OneNet_DevLink(void) {
    MQTT_PACKET_STRUCTURE mqttPacket = {NULL, 0, 0, 0};
    unsigned char *dataPtr;
    _Bool status = 1;
    
    // 1. 先关闭可能存在的旧连接
    ESP8266_CloseTCP();
    Delay_ms(200);
    
    // 2. 建立 TCP 连接到 OneNET 服务器
    if(!ESP8266_StartTCP(ONENET_SERVER_IP, ONENET_SERVER_PORT)) {
        return 1;  // TCP 连接失败
    }
    Delay_ms(200);
    
    // 3. 发送 MQTT CONNECT 包
    if(MQTT_PacketConnect(PROID, AUTH_INFO, DEVID, 256, 1, 
                          MQTT_QOS_LEVEL0, NULL, NULL, 0, &mqttPacket) == 0) {
        ESP8266_SendData(mqttPacket._data, mqttPacket._len);
        
        dataPtr = ESP8266_GetIPD(1000);
        if(dataPtr != NULL) {
            if(MQTT_UnPacketRecv(dataPtr) == MQTT_PKT_CONNACK) {
                switch(MQTT_UnPacketConnectAck(dataPtr)) {
                    case 0: status = 0; break;  // 连接成功
                    default: break;
                }
            }
        }
        
        MQTT_DeleteBuffer(&mqttPacket);
    }
    
    return status;
}
```

// 构造JSON数据
unsigned char MqttOnenet_Savedata(char *t_payload) {
    char json[] = "{\"id\":\"123\",\"version\":\"1.0\",\"msgId\":\"123\","
                  "\"params\":{"
                  "\"temperature\":{\"value\":%d},"
                  "\"humidity\":{\"value\":%d},"

// 上传数据到OneNET
uint8_t OneNet_SendData(void) {
    MQTT_PACKET_STRUCTURE mqttPacket = {NULL, 0, 0, 0};
    char payload[256];
    unsigned short payload_len = 0;
    
    // 构造数据
    payload_len = MqttOnenet_Savedata(payload);
    
    // 打包MQTT PUBLISH包
    if(MQTT_PacketPublish("$sys/"PROID"/"DEVID"/dp/post/plain", 
                          payload, payload_len, &mqttPacket) == 0) {
        ESP8266_SendData(mqttPacket._data, mqttPacket._len);
        MQTT_DeleteBuffer(&mqttPacket);
        return 0;
    }
    
    return 1;
}

## 七、常见问题与解决方案

### 7.1 LED 不亮

**可能原因**：
1. 硬件连接问题
2. GPIO 配置错误
3. 高低电平逻辑反了

**解决方案**：
- 确认野火霸道板 LED 为**低电平点亮**
- 检查 GPIO 是否配置为推挽输出
- 尝试使用 BSRR 寄存器操作

### 7.2 按键无反应

**可能原因**：
1. 输入模式配置错误
2. 缺少上拉电阻
3. PA0 唤醒功能干扰

**解决方案**：
- 使用 GPIO_Mode_IPU（上拉输入）或浮空输入
- 野火霸道板 KEY1 有外部上拉，可使用浮空输入
- 避免使用 PC13（有限制）

### 7.3 串口乱码

**可能原因**：
1. 波特率不匹配
2. 系统时钟配置错误

**解决方案**：
```c
// 72MHz 时的 USART1 BRR 值
USART1->BRR = 0x0271;   // 115200 @ 72MHz

// 8MHz HSI 时的 USART1 BRR 值
USART1->BRR = 0x0045;   // 115200 @ 8MHz
```

### 7.4 ESP8266 连接失败

**可能原因**：
1. 接线错误（TX/RX 反了）
2. 供电不足（需要 3.3V，不是 5V）
3. CH_PD 未拉高
4. 波特率不匹配
5. WiFi名称或密码错误
6. WiFi信号太弱

**解决方案**：
- 确认 TX/RX 交叉连接
- 使用 3.3V 供电
- CH_PD 接 3.3V
- ESP8266 默认波特率 115200
- 检查WiFi信息是否正确
- 确保WiFi信号良好

### 7.5 MQTT 连接失败

**可能原因**：
1. 产品ID、设备名称、Token错误
2. TCP连接未建立成功
3. 缓冲区太小，MQTT包被截断

**解决方案**：
- 确认 OneNET 平台配置正确
- 检查 TCP 连接是否成功
- 增大接收缓冲区到 256 字节以上
